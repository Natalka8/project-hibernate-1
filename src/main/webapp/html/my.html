function showList(page_number) { // It should be an AJAX request to the REST API, not static data. let url = "../rest/players?"; let countPerPage = $("#count_1").val(); if (countPerPage === null) { countPerPage = 3; } url = url.concat("pageSize=").concat(countPerPage); if (page_number !== null) { url = url.concat("&pageNumber=").concat(page_number); } $.get(url, function (response) { // Clearing the table $("tr:has(td)").remove(); //Adding data from the REST API $.each(response, function (i, item) { $('').html("" + item.id + "" + item.name + "" + item.title + "" + item.race + "" + item.profession + "" + item.level + "" + new Date(item.birthday).toLocaleDateString() + "" + item.banned + "" + "" + "") .appendTo('#table_1'); }); }); updatePagination(page_number); }
